
#include <iostream>
#include <limits>
using namespace std;

// Global 3x3 game board
char board[3][3] = {
    {'1', '2', '3'},
    {'4', '5', '6'},
    {'7', '8', '9'}
};

// Function to display the game board
void displayBoard() {

    cout << "\n";

    for (int i = 0; i < 3; i++) {

        cout << " ";

        for (int j = 0; j < 3; j++) {

            cout << board[i][j];

            if (j < 2)
                cout << " | ";
        }

        cout << endl;

        if (i < 2)
            cout << "---+---+---" << endl;
    }

    cout << "\n";
}

// Function to check winning conditions
bool checkWin(char player) {

    // Check rows
    for (int i = 0; i < 3; i++) {

        if (board[i][0] == player &&
            board[i][1] == player &&
            board[i][2] == player) {

            return true;
        }
    }

    // Check columns
    for (int j = 0; j < 3; j++) {

        if (board[0][j] == player &&
            board[1][j] == player &&
            board[2][j] == player) {

            return true;
        }
    }

    // Check first diagonal
    if (board[0][0] == player &&
        board[1][1] == player &&
        board[2][2] == player) {

        return true;
    }

    // Check second diagonal
    if (board[0][2] == player &&
        board[1][1] == player &&
        board[2][0] == player) {

        return true;
    }

    return false;
}

// Function to check draw condition
bool checkDraw() {

    for (int i = 0; i < 3; i++) {

        for (int j = 0; j < 3; j++) {

            if (board[i][j] != 'X' &&
                board[i][j] != 'O') {

                return false;
            }
        }
    }

    return true;
}

// Function to accept and validate player move
bool makeMove(char player) {

    int position;

    cout << "Player " << player
         << ", enter position (1-9): ";

    if (!(cin >> position)) {

        cout << "Invalid input! Enter a number.\n";

        cin.clear();

        cin.ignore(
            numeric_limits<streamsize>::max(), '\n'
        );

        return false;
    }

    if (position < 1 || position > 9) {

        cout << "Invalid position! Choose 1-9.\n";

        return false;
    }

    // Convert position into row and column
    int row = (position - 1) / 3;
    int col = (position - 1) % 3;

    // Check whether the cell is already occupied
    if (board[row][col] == 'X' ||
        board[row][col] == 'O') {

        cout << "Position already occupied! Try again.\n";

        return false;
    }

    // Place player's symbol
    board[row][col] = player;

    return true;
}

// Main game function
void playGame() {

    char currentPlayer = 'X';

    bool gameOver = false;

    cout << "============================\n";
    cout << "     TIC TAC TOE GAME\n";
    cout << "============================\n";

    cout << "\nPlayer 1: X";
    cout << "\nPlayer 2: O\n";

    while (!gameOver) {

        displayBoard();

        // Repeat the same player's turn if move is invalid
        if (!makeMove(currentPlayer)) {
            continue;
        }

        displayBoard();

        // Check winner
        if (checkWin(currentPlayer)) {

            cout << "Congratulations!\n";
            cout << "Player " << currentPlayer
                 << " wins the game!\n";

            gameOver = true;
        }

        // Check draw
        else if (checkDraw()) {

            cout << "Game Draw!\n";

            gameOver = true;
        }

        // Switch player
        else {

            if (currentPlayer == 'X') {
                currentPlayer = 'O';
            }
            else {
                currentPlayer = 'X';
            }
        }
    }

    cout << "\n===== GAME OVER =====\n";
}

// Main function
int main() {

    playGame();

    return 0;
}
